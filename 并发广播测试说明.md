# 并发广播测试说明

本文档单独说明第六章 `图6-4 并发广播测试日志图` 要怎么测、用什么命令、看什么结果、截哪里。

注意：下面这些命令要在你的**实际项目根目录**执行，不是在当前 `lunwen` 论文目录执行。

## 一、测试目标

这一组测试对应论文第六章“并发性能测试”：

- 建立 `20`、`50`、`100` 个 Socket 客户端
- 所有客户端加入同一个聊天室
- 由其中一个客户端触发一条群消息广播
- 统计所有客户端收到该消息的时延
- 输出平均值、P95、最大值和成功率

## 二、先准备什么

先从你已经能正常访问的 Web 端里拿到下面 3 个值：

- `ROOM_ID`
- `USER_ID`
- `TOKEN`

### 1. 找 `ROOM_ID`

打开聊天室页面后按 `F12`：

1. 进入 `Network`
2. 选择 `WS` 或 `Socket`
3. 找到 `join-group`
4. 记录里面的 `roomId`

### 2. 找 `USER_ID`

打开 `Application -> Local Storage`，找到当前登录用户信息，记录：

- `userId`

### 3. 找 `TOKEN`

同样在 `Local Storage` 或登录返回里找到：

- `token`

后面命令都要把这 3 个值替换进去。

## 三、先启动后端


## 四、单轮广播测试命令

开第二个 PowerShell 窗口，进入项目根目录后执行。

先测 `100` 客户端，`20` 和 `50` 只需要改 `CLIENTS` 的值再跑一次。

```powershell
cd F:\你的项目根目录
$env:ROOM_ID="chatroom_xxx"
$env:USER_ID="1742902338123"
$env:TOKEN="你的BearerToken"
$env:CLIENTS="100"
@'
const { io } = require("socket.io-client");

const baseUrl = "http://localhost:3000";
const roomId = process.env.ROOM_ID;
const userId = process.env.USER_ID;
const token = process.env.TOKEN;
const total = Number(process.env.CLIENTS || 100);
const mark = `ch6_broadcast_${Date.now()}`;
const sockets = [];
const received = new Set();
const delays = [];
let joined = 0;
let startedAt = 0;
let finished = false;

function percentile(arr, p) {
  const s = [...arr].sort((a, b) => a - b);
  const idx = Math.ceil(s.length * p) - 1;
  return s[Math.max(0, idx)];
}

function finish() {
  if (finished) return;
  finished = true;
  if (delays.length === 0) {
    console.log("===== RESULT =====");
    console.log(`clients=${total}`);
    console.log(`roomId=${roomId}`);
    console.log(`received=0/${total}`);
    console.log("successRate=0%");
    console.log(`mark=${mark}`);
    setTimeout(() => sockets.forEach(s => s.close()), 800);
    return;
  }
  const avg = Math.round(delays.reduce((a, b) => a + b, 0) / delays.length);
  const p95 = percentile(delays, 0.95);
  const max = Math.max(...delays);
  console.log("===== RESULT =====");
  console.log(`clients=${total}`);
  console.log(`roomId=${roomId}`);
  console.log(`received=${received.size}/${total}`);
  console.log(`successRate=${(received.size / total * 100).toFixed(0)}%`);
  console.log(`avg=${avg}ms`);
  console.log(`p95=${p95}ms`);
  console.log(`max=${max}ms`);
  console.log(`mark=${mark}`);
  setTimeout(() => sockets.forEach(s => s.close()), 800);
}

for (let i = 0; i < total; i++) {
  const socket = io(baseUrl, { transports: ["websocket", "polling"] });
  sockets.push(socket);

  socket.on("connect", () => {
    socket.emit("login", userId);
    socket.emit("join-group", { roomId, userId });
    joined++;
    console.log(`[join ${joined}/${total}] socket=${socket.id}`);
    if (joined === total) {
      setTimeout(async () => {
        startedAt = Date.now();
        console.log(`SEND start=${startedAt} content=${mark}`);
        const res = await fetch(`${baseUrl}/room/${roomId}/messages`, {
          method: "POST",
          headers: {
            "Content-Type": "application/json",
            "Authorization": `Bearer ${token}`
          },
          body: JSON.stringify({
            content: mark,
            messageType: "text"
          })
        });
        console.log(`HTTP status=${res.status}`);
      }, 1000);
    }
  });

  socket.on("group-message", (msg) => {
    if (!msg || msg.content !== mark) return;
    if (received.has(socket.id)) return;
    received.add(socket.id);
    delays.push(Date.now() - startedAt);
    console.log(`[recv ${received.size}/${total}] ${delays[delays.length - 1]}ms socket=${socket.id}`);
    if (received.size === total) finish();
  });

  socket.on("connect_error", (err) => {
    console.log(`[error] ${err.message}`);
  });

  socket.on("disconnect", (reason) => {
    console.log(`[disconnect] socket=${socket.id} reason=${reason}`);
  });
}

setTimeout(() => {
  if (!finished) finish();
}, 30000);
'@ | node -
```

## 六、100 客户端连续 10 轮广播命令

这个对应论文里的 `CON-04`。

```powershell
cd F:\你的项目根目录
$env:ROOM_ID="chatroom_xxx"
$env:USER_ID="1742902338123"
$env:TOKEN="你的BearerToken"
$env:CLIENTS="100"
$env:ROUNDS="10"
@'
const { io } = require("socket.io-client");

const baseUrl = "http://localhost:3000";
const roomId = process.env.ROOM_ID;
const userId = process.env.USER_ID;
const token = process.env.TOKEN;
const total = Number(process.env.CLIENTS || 100);
const rounds = Number(process.env.ROUNDS || 10);
const sockets = [];
let joined = 0;
let disconnects = 0;
let errors = 0;
let totalDelivered = 0;
const state = new Map();

async function sendRound(round) {
  const mark = `ch6_round_${round}_${Date.now()}`;
  state.set(mark, new Set());
  console.log(`ROUND ${round}/${rounds} SEND ${mark}`);
  const res = await fetch(`${baseUrl}/room/${roomId}/messages`, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "Authorization": `Bearer ${token}`
    },
    body: JSON.stringify({
      content: mark,
      messageType: "text"
    })
  });
  console.log(`ROUND ${round}/${rounds} HTTP status=${res.status}`);
}

function allRoundsDone() {
  const delivered = Array.from(state.values()).reduce((sum, set) => sum + set.size, 0);
  return delivered >= total * rounds;
}

function finish() {
  const delivered = Array.from(state.values()).reduce((sum, set) => sum + set.size, 0);
  console.log("===== RESULT =====");
  console.log(`clients=${total}`);
  console.log(`rounds=${rounds}`);
  console.log(`deliveries=${delivered}/${total * rounds}`);
  console.log(`successRate=${(delivered / (total * rounds) * 100).toFixed(0)}%`);
  console.log(`disconnects=${disconnects}`);
  console.log(`errors=${errors}`);
  setTimeout(() => sockets.forEach(s => s.close()), 800);
}

for (let i = 0; i < total; i++) {
  const socket = io(baseUrl, { transports: ["websocket", "polling"] });
  sockets.push(socket);

  socket.on("connect", async () => {
    socket.emit("login", userId);
    socket.emit("join-group", { roomId, userId });
    joined++;
    console.log(`[join ${joined}/${total}] socket=${socket.id}`);
    if (joined === total) {
      for (let round = 1; round <= rounds; round++) {
        await new Promise(r => setTimeout(r, 800));
        await sendRound(round);
      }
    }
  });

  socket.on("group-message", (msg) => {
    if (!msg || !msg.content || !msg.content.startsWith("ch6_round_")) return;
    if (!state.has(msg.content)) return;
    const set = state.get(msg.content);
    if (set.has(socket.id)) return;
    set.add(socket.id);
    totalDelivered++;
    console.log(`[recv ${totalDelivered}/${total * rounds}] socket=${socket.id} content=${msg.content}`);
    if (allRoundsDone()) finish();
  });

  socket.on("connect_error", (err) => {
    errors++;
    console.log(`[error] ${err.message}`);
  });

  socket.on("disconnect", (reason) => {
    disconnects++;
    console.log(`[disconnect] socket=${socket.id} reason=${reason}`);
  });
}

setTimeout(() => {
  finish();
}, 60000);
'@ | node -
```

## 七、日志结果怎么看

你主要看脚本最后汇总出来的这些字段：

### 单轮广播

```text
clients=100
received=100/100
successRate=100%
avg=4ms
p95=5ms
max=5ms
```

含义如下：

- `clients=100`：这次测试建了 100 个客户端
- `received=100/100`：100 个客户端都收到了广播
- `successRate=100%`：送达成功率 100%
- `avg`：平均时延
- `p95`：95 分位时延
- `max`：最大时延

### 连续 10 轮广播

```text
clients=100
rounds=10
deliveries=1000/1000
successRate=100%
disconnects=0
errors=0
```

含义如下：

- `deliveries=1000/1000`：100 个客户端 * 10 轮 = 1000 次投递全部成功
- `disconnects=0`：没有异常断连
- `errors=0`：没有明显错误

## 十、如果跑不起来先检查什么

如果结果不对，优先检查这几项：

1. `ROOM_ID` 是否正确
2. `TOKEN` 是否过期
3. `USER_ID` 是否和当前登录用户一致
4. `http://localhost:3000` 是否能访问
5. 项目是否真的使用 `join-group` 和 `group-message`
